\question[10]\mbox{}
\textbf{TRUE or FALSE!}  Write  TRUE if the statement is true.  Otherwise, write FALSE.  Your response should be in ALL CAPS.  No justification is required.
\begin{enumerate}[(a)]
\item  The group $\mathbb Q$ with binary operation $+$ is a cyclic group.
\vspace{1.3in}
\item  Every group of order $7$ is cyclic.
\vspace{1.3in}
\item  If $G$ is non-Abelian and $H\leq G$ then $H$ is also non-Abelian
\vspace{1.3in}
\item  Every nontrivial group has a nontrivial proper subgroup.
\vspace{1.3in}
\item  If $G$ is a group and $x,y\in G$ then $x^3 = y^3$ if and only if $x=y$
\vspace{1.3in}
\end{enumerate}

\newpage
\question[10]\mbox{}
\begin{enumerate}[(a)]
\item  State Lagrange's Theorem
\vspace{2in}
\item  Write the definition of a homomorphism from a group $G$ to a group $H$.
\vspace{2in}
\item  Give an example of a group of order $6$ which is not Abelian.
\end{enumerate}

\newpage
\question[10]\mbox{}
Let $G$ be a group and $\varnothing\neq H\subseteq G$.
Prove that $H$ is a subgroup if and only if $ab^{-1}\in H$ for all $a,b\in H$.

\newpage
\question[10]\mbox{}
Consider the set
$$G = (\mathbb{Z}_7)^\times\times \mathbb{Z}_7 = \{(a,b): a,b\in\mathbb{Z}_7,\ a\neq 0\},$$
with the binary operation
$$(a,b) \ast (c,d) = (a\times_7c,b+_7d),$$
where here $\times_7$ and $+_7$ are multiplication and addition mod $7$.

\begin{enumerate}[(a)]
\item Show that $G$ has an identity element
\vspace{1.5in}
\item Show each element of $G$ has an inverse
\vspace{2in}
\item Prove that $G$ is a cyclic group
\vspace{2in}
\item Explain why $G$ is isomorphic to $\mathbb{Z}_{42}$
\end{enumerate}

\newpage
\question[10]\mbox{}
Let $H$ and $K$ be subgroups of a group $G$.
\begin{enumerate}[(a)]
\item Prove that $H\cap K$ is a subgroup.
\vspace{2in}
\item Prove that if $K$ is a normal subgroup of $G$, then $H\cap K$ is a normal subgroup of $H$.
\vspace{3in}
\item Prove that if $H$ is also a normal subgroup of $G$, then $H\cap K$ is a normal subgroup of $G$.
\end{enumerate}

\newpage
\question[10]\mbox{}
Consider the group $G=S_4$ and the subgroup $H=\langle (1234)\rangle$.
\begin{enumerate}[(a)]
\item Explicitly describe all the elements in $H$.
\vspace{2in}
\item Explicitly write out all of the left cosets of $H$ in $G$.  Carefully explain why what list are \emph{all} the left cosets.
\end{enumerate}

\newpage
\question[10]\mbox{}

Suppose that $G$ is a finite group (possibly non-Abelian).  Show that if $\varphi$ is a homomorphism

$$\varphi: G\rightarrow\mathbb Z$$

then $\varphi(x) = 0$ for all $x\in G$.  In other words, the only homomorphism from $G$ to $\mathbb Z$ is the trivial one.
